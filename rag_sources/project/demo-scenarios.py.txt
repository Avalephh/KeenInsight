title: str
description: str
query: str | None
parameters: dict[str, Any]

def get_scenario(name: str) -> dict[str, Any]:
for scenario in SCENARIO_METADATA:
if scenario["name"] == name:
return scenario
raise KeyError(f"unknown workload scenario: {name}")
