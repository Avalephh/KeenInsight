<script>
(() => {
"use strict";
const message = document.getElementById("bootMessage");
const detail = document.getElementById("bootDetail");
const progress = document.getElementById("bootProgress");
const retry = document.getElementById("retryButton");
const hash = window.location.hash || "";
let timer = null;
let inFlight = false;
let redirected = false;

function step(name, state, text) {
const node = document.querySelector(`[data-step="${name}"]`);
if (!node) return;
node.dataset.state = state;
node.querySelector(".step-state").textContent = text;
if (state === "done") node.querySelector(".step-icon").textContent = "✓";
}

function schedule() {
clearTimeout(timer);
timer = setTimeout(check, 1000);
}

async function check() {
if (inFlight || redirected) return;
inFlight = true;
retry.classList.add("hidden");
step("service", "active", "检查中");
try {
const healthResponse = await fetch("/api/health", { cache: "no-store", headers: { "Cache-Control": "no-cache" } });
if (!healthResponse.ok) throw new Error(`服务响应 ${healthResponse.status}`);
const health = await healthResponse.json();
step("service", "done", "已响应");
if (!health.ok || !health.database) {
step("database", "active", "重试中");
step("telemetry", "pending", "等待中");
step("overview", "pending", "等待中");
message.textContent = "PostgreSQL 还没有完全就绪，正在自动重试…";
detail.textContent = "连接恢复后会自动进入总览。";
progress.style.width = "25%";
schedule();
return;
}
step("database", "done", "已连接");
const collector = health.collector || {};
if (!collector.running || !collector.first_sample_ready) {
step("telemetry", "active", "采集中");
step("overview", "pending", "等待中");
message.textContent = "数据库已连接，正在等待首条真实监控快照…";
detail.textContent = "后台采集器会持续读取当前实例的指标。";
progress.style.width = "50%";
schedule();
return;
}
step("telemetry", "done", "已采集");
const [dashboardResponse, metricsResponse] = await Promise.all([
fetch("/api/dashboard", { cache: "no-store", headers: { "Cache-Control": "no-cache" } }),
fetch("/api/dashboard/metrics?limit=1", { cache: "no-store", headers: { "Cache-Control": "no-cache" } }),
]);
if (!dashboardResponse.ok) throw new Error(`总览响应 ${dashboardResponse.status}`);
if (!metricsResponse.ok) throw new Error(`指标响应 ${metricsResponse.status}`);
const dashboard = await dashboardResponse.json();
const metrics = await metricsResponse.json();
if (!dashboard.snapshot || !(metrics.items || []).length) {
step("overview", "active", "准备中");
message.textContent = "实时快照已开始采集，正在准备总览页面…";
detail.textContent = "马上进入主页。";
progress.style.width = "75%";
schedule();
return;
}
step("overview", "done", "已就绪");
progress.style.width = "100%";
message.textContent = "连接完成，正在进入 PostgreSQL 总览…";
detail.textContent = `最近观测：${new Date(dashboard.snapshot.collected_at).toLocaleString()}`;
redirected = true;
window.setTimeout(() => window.location.replace(`/console/home${hash}`), 180);
} catch (error) {
step("service", "error", "等待重试");
step("database", "pending", "等待中");
step("telemetry", "pending", "等待中");
step("overview", "pending", "等待中");
message.textContent = "控制面暂时无法完成连接，正在后台重试…";
detail.textContent = "如果网络恢复，页面会自动继续；也可以手动重试。";
retry.classList.remove("hidden");
progress.style.width = "10%";
schedule();
} finally {
inFlight = false;
}
}

retry.addEventListener("click", check);
check();
})();
</script>
