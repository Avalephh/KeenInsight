const pretty = (value) => JSON.stringify(value, null, 2);
const fmt = (value, digits = 2) => {
if (value === null || value === undefined || value === "") return "—";
const number = Number(value);
return Number.isFinite(number) ? number.toLocaleString(undefined, { maximumFractionDigits: digits }) : esc(value);
};
const formatBytes = (value) => {
const number = Number(value);
if (!Number.isFinite(number)) return "—";
const units = ["B", "KiB", "MiB", "GiB", "TiB"];
let amount = number, index = 0;
while (amount >= 1024 && index < units.length - 1) { amount /= 1024; index += 1; }
return `${amount.toLocaleString(undefined, { maximumFractionDigits: 1 })} ${units[index]}`;
};
const date = (value) => value ? new Date(value).toLocaleString() : "—";
const locatorText = (value) => {
const locator = value || {};
if (!Object.keys(locator).length) return "";
const pages = locator.page_start ? (locator.page_end && locator.page_end !== locator.page_start ? `pages ${locator.page_start}–${locator.page_end}` : `page ${locator.page_start}`) : "";
const lines = locator.line_start ? `normalized lines ${locator.line_start}–${locator.line_end || locator.line_start}` : "";
return [pages, lines].filter(Boolean).join(" · ");
};

}

function drawChart(items) {
const canvas = $("metricsCanvas"), ctx = canvas.getContext("2d"), ratio = window.devicePixelRatio || 1, width = canvas.clientWidth || 600, height = 250;
canvas.width = width * ratio; canvas.height = height * ratio; ctx.setTransform(ratio, 0, 0, ratio, 0, 0); ctx.clearRect(0, 0, width, height);
const series = [{ key: "xact_commit", label: "commits/s", color: "#2457d6" }, { key: "xact_rollback", label: "rollbacks/s", color: "#d33a4c" }, { key: "temp_bytes", label: "temp bytes/s", color: "#b96b00" }];
const values = series.map((s) => items.map((item) => Number(item.payload?.rates?.counters_per_second?.[s.key] || 0)));
const max = Math.max(1, ...values.flat()); const pad = { left: 35, right: 12, top: 12, bottom: 25 }; const chartW = width - pad.left - pad.right, chartH = height - pad.top - pad.bottom;
ctx.strokeStyle = "#e9edf3"; ctx.lineWidth = 1; ctx.font = "10px system-ui"; ctx.fillStyle = "#8b96a6";
for (let i = 0; i <= 4; i++) { const y = pad.top + chartH * i / 4; ctx.beginPath(); ctx.moveTo(pad.left, y); ctx.lineTo(width - pad.right, y); ctx.stroke(); ctx.fillText(fmt(max * (1 - i / 4), 1), 2, y + 3); }
series.forEach((s, si) => { if (!values[si].length) return; ctx.strokeStyle = s.color; ctx.lineWidth = 2; ctx.beginPath(); values[si].forEach((v, i) => { const x = pad.left + (values[si].length === 1 ? chartW / 2 : chartW * i / (values[si].length - 1)); const y = pad.top + chartH * (1 - v / max); i ? ctx.lineTo(x, y) : ctx.moveTo(x, y); }); ctx.stroke(); });
$("chartLegend").innerHTML = series.map((s) => `<span class="legend-item"><span class="legend-color" style="background:${s.color}"></span>${s.label}</span>`).join("");
}

function rememberRun(result) {
state.lastRun = result;
state.guidedCurrent[result.scenario] = true;
if (result.phase === "baseline") {
state.baselineRuns[result.scenario] = result.run_id;
state.baselineResults[result.scenario] = result;
state.actionsByScenario[result.scenario] = undefined;
state.actionRecords[result.scenario] = undefined;
state.feedbackByScenario[result.scenario] = undefined;
state.afterRuns[result.scenario] = undefined;
state.afterResults[result.scenario] = undefined;
}
if (result.phase === "after") {
state.afterRuns[result.scenario] = result.run_id;
state.afterResults[result.scenario] = result;
}
}

async function runWorkloadPhase(scenarioName, phase) {
const result = await api("/api/workloads/" + scenarioName + "/runs", {
method: "POST",
body: JSON.stringify({
phase,
repetitions: workloadRepetitions(),
warmup_repetitions: workloadWarmups(),
parameters: readWorkloadParameters(scenarioName),
}),
});
rememberRun(result);
showWorkloadResult(result);
toast(phase + " measurement completed from PostgreSQL");
await loadWorkloadRuns();
await refreshDashboard();
renderGuidedDemo();
return result;
}

function rememberActionRecords(items) {
(items || []).forEach((row) => {
const source = row.validation?.proposal_source;
if (source?.type !== "workload_run" || !source.scenario) return;
const currentBaseline = state.baselineRuns[source.scenario];
if (currentBaseline !== undefined && Number(currentBaseline) !== Number(source.run_id)) return;
const current = state.actionRecords[source.scenario];
if (!current || Number(row.id) > Number(current.id)) {
state.actionRecords[source.scenario] = row;
}
if (row.validation?.latest_feedback) {
state.feedbackByScenario[source.scenario] = row.validation.latest_feedback;
}
});
}

function updateActionRecord(action) {
const source = action?.validation?.proposal_source;
if (!source?.scenario) return;
state.actionsByScenario[source.scenario] = action.id;
state.actionRecords[source.scenario] = action;
}

async function runGuidedStep(step) {
if (state.guidedBusy) return;
const scenario = state.guidedScenario;
state.guidedBusy = true;
renderGuidedDemo();
try {
if (step === "baseline") {
await runWorkloadPhase(scenario, "baseline");
} else if (step === "propose") {
const action = await api("/api/workload-runs/" + state.baselineRuns[scenario] + "/propose", { method: "POST" });
updateActionRecord(action);
state.lastRun = state.baselineResults[scenario];
showWorkloadResult(state.lastRun);
await loadActions();
toast("Action #" + action.id + " proposed for review");
} else if (step === "apply") {
const action = await api("/api/actions/" + state.actionsByScenario[scenario] + "/apply", { method: "POST" });
updateActionRecord(action);
await loadActions();
toast("Reviewed tuning action applied and verified in PostgreSQL");
} else if (step === "after") {
await runWorkloadPhase(scenario, "after");
} else if (step === "feedback") {
const data = await api("/api/actions/" + state.actionsByScenario[scenario] + "/feedback", {
method: "POST",
body: JSON.stringify({
baseline_run_id: Number(state.baselineRuns[scenario]),
after_run_id: Number(state.afterRuns[scenario]),
}),
});
state.feedbackByScenario[scenario] = data.feedback?.outcome || {};
updateActionRecord(data.action);
await loadActions();
await loadKnowledge();
showWorkloadResult(state.afterResults[scenario]);
toast("Measured outcome recorded: " + (data.feedback?.outcome?.observed_effect || "complete"));
} else if (step === "rollback") {
const action = await api("/api/actions/" + state.actionsByScenario[scenario] + "/rollback", { method: "POST" });
updateActionRecord(action);
await loadActions();
await refreshDashboard();
toast("Tuning action rolled back and PostgreSQL state verified");
}
} catch (e) {
toast(e.message, true);
} finally {
state.guidedBusy = false;
renderGuidedDemo();
}
}

async function autoResolveGuided() {
if (state.guidedBusy) return;
const scenario = state.guidedScenario;
const baselineRunId = state.baselineRuns[scenario];
if (!baselineRunId) return toast("Run a baseline before automatic resolution", true);
if (state.actionRecords[scenario]) return toast("This baseline already has a proposal; start a fresh baseline", true);
if (!confirm("Automatically apply the evidence-backed action, measure the same SQL, record feedback, and restore it after verification?")) return;
state.guidedBusy = true;
renderGuidedDemo();
try {
const data = await api(`/api/workload-runs/${baselineRunId}/auto-resolve`, {
method: "POST",
body: JSON.stringify({ rollback_after_success: Boolean($("guidedAutoRollback")?.checked) }),
});
const action = data.action || {};
const after = data.after || {};
const feedback = data.feedback?.feedback?.outcome || {};
state.guidedCurrent[scenario] = true;
state.baselineRuns[scenario] = Number(data.baseline?.id || baselineRunId);
state.baselineResults[scenario] = { ...(data.baseline || {}), run_id: state.baselineRuns[scenario], scenario, phase: "baseline", observations: data.baseline?.observations || {} };
state.actionsByScenario[scenario] = action.id;
state.actionRecords[scenario] = action;
state.afterRuns[scenario] = Number(after.run_id);
state.afterResults[scenario] = after;
state.feedbackByScenario[scenario] = feedback;
state.lastRun = after;
showWorkloadResult(after);
await Promise.all([loadWorkloadRuns(), loadActions(), loadKnowledge(), refreshDashboard()]);
toast(`Automatic loop completed: ${data.resolution?.effective ? "effective" : "not effective"}${data.resolution?.rollback_performed ? "; state restored" : "; action deployed"}`);
} catch (e) { toast(e.message, true); }
finally { state.guidedBusy = false; renderGuidedDemo(); }
}

function formatPercent(value) {

}

function planEvidence(observations) {
const summary = observations?.plan_summary || {};
const plan = observations?.plan || {};
const nodes = plan.nodes || [];
const hashes = nodes.filter((node) => node["Node Type"] === "Hash").map((node) => node["Hash Batches"] ?? node.Batches).filter((value) => value !== undefined);
const indexes = nodes.map((node) => node["Index Name"]).filter(Boolean);
const sorts = nodes.filter((node) => node["Sort Method"]).map((node) => {
let value = String(node["Sort Method"]);
if (node["Sort Space Type"]) value += " / " + node["Sort Space Type"];
if (node["Sort Space Used"]) value += " (" + node["Sort Space Used"] + " kB)";
return value;
});
const aggregates = nodes.filter((node) => ["Aggregate", "HashAggregate"].includes(node["Node Type"]) && node.Strategy).map((node) => String(node["Node Type"]) + ": " + String(node.Strategy));
const workersPlanned = Math.max(0, ...nodes.map((node) => Number(node["Workers Planned"] || 0)));
const workersLaunched = Math.max(0, ...nodes.map((node) => Number(node["Workers Launched"] || 0)));
return `<div class="evidence-grid">
<div class="evidence-item"><span>Root node</span><code>${esc(summary.root_node || "—")}</code></div>
<div class="evidence-item"><span>Scan nodes</span><code>${esc((summary.scan_nodes || []).join(" → ") || "—")}</code></div>
<div class="evidence-item"><span>Join nodes</span><code>${esc((summary.join_nodes || []).join(" → ") || "—")}</code></div>
<div class="evidence-item"><span>Hash batches</span><code>${esc(hashes.length ? hashes.join(", ") : "not observed")}</code></div>
<div class="evidence-item"><span>Sort method</span><code>${esc(sorts.join(" → ") || "not observed")}</code></div>
<div class="evidence-item"><span>Aggregate strategy</span><code>${esc(aggregates.join(" → ") || "not observed")}</code></div>
<div class="evidence-item"><span>Temporary blocks</span><code>${summary.has_temp_blocks ? "observed" : "not observed"}</code></div>
<div class="evidence-item"><span>Indexes used</span><code>${esc(indexes.join(", ") || "none observed")}</code></div>
<div class="evidence-item"><span>Workers planned</span><code>${fmt(workersPlanned, 0)}</code></div>

function comparisonEvidence(result) {
const baselineResult = state.baselineResults[result.scenario];
const baseline = baselineResult?.observations || baselineResult;
const after = result.observations || result;
if (!baseline || result.phase !== "after") return "";
const baseMedian = Number(baseline.median_query_time_ms);
const afterMedian = Number(after?.median_query_time_ms);
const baseP95 = Number(baseline.p95_query_time_ms);
const afterP95 = Number(after?.p95_query_time_ms);
const improvement = (before, current) => before > 0 && Number.isFinite(before) && Number.isFinite(current) ? ((before - current) / before) * 100 : null;
const medianImprovement = improvement(baseMedian, afterMedian);
const p95Improvement = improvement(baseP95, afterP95);
const recorded = state.feedbackByScenario[result.scenario];
const status = recorded
? `<span class="badge ${recorded.effective ? "applied" : "warning"}">${esc(recorded.observed_effect || "recorded")}</span>`
: `<span class="badge info">awaiting server feedback</span>`;
return `<div class="comparison-panel">
<div class="comparison-header"><div><span class="eyebrow">Measured comparison</span><h3>Baseline → after</h3></div>${status}</div>
<div class="comparison-grid">
${workloadMetric("Median latency", `${fmt(baseMedian)} → ${fmt(afterMedian)} ms`, `observed improvement ${formatPercent(medianImprovement)}`, medianImprovement !== null && medianImprovement > 0 ? "good" : "bad")}
${workloadMetric("P95 latency", `${fmt(baseP95)} → ${fmt(afterP95)} ms`, `observed improvement ${formatPercent(p95Improvement)}`, p95Improvement !== null && p95Improvement > 0 ? "good" : "bad")}
</div>
<p class="tiny muted">This preview uses the two returned measurements. The server records <code>effective</code> only after plan, result, parameter, and repetition checks.</p>
</div>`;
}

function planSignals(observations) {
const summary = observations?.plan_summary || {};
const nodes = observations?.plan?.nodes || [];
const hashes = nodes
.filter((node) => node["Node Type"] === "Hash")
.map((node) => node["Hash Batches"] ?? node.Batches)
.filter((value) => value !== undefined);
const sorts = summary.sort_methods || nodes
.filter((node) => node["Sort Method"])
.map((node) => node["Sort Method"]);
const sortSpaceTypes = summary.sort_space_types || nodes
.filter((node) => node["Sort Space Type"])
.map((node) => node["Sort Space Type"]);
const aggregates = summary.aggregate_strategies || nodes
.filter((node) => ["Aggregate", "HashAggregate"].includes(node["Node Type"]) && node.Strategy)
.map((node) => node.Strategy);
const temporary = Boolean(summary.has_temp_blocks) || nodes.some((node) => (
Number(node["Temp Read Blocks"] || 0) > 0 || Number(node["Temp Written Blocks"] || 0) > 0
));
return {
root: summary.root_node || observations?.plan?.root?.["Node Type"] || "not observed",
scans: (summary.scan_nodes || []).join(" → ") || "not observed",
joins: (summary.join_nodes || []).join(" → ") || "not observed",
hashBatches: hashes.length ? hashes.join(", ") : "not observed",
sortMethods: sorts.length ? sorts.join(", ") : "not observed",
sortSpaceTypes: sortSpaceTypes.length ? sortSpaceTypes.join(", ") : "not observed",
aggregates: aggregates.length ? aggregates.join(", ") : "not observed",
temporary: temporary ? "observed" : "not observed",
workers: Math.max(0, ...nodes.map((node) => Number(node["Workers Launched"] || 0))),
plannedWorkers: Math.max(0, ...nodes.map((node) => Number(node["Workers Planned"] || 0))),
indexes: nodes.map((node) => node["Index Name"]).filter(Boolean).join(", ") || "not observed",
estimateError: observations?.statistics_measurement?.estimate_error_ratio === undefined ? "not observed" : `${Number(observations.statistics_measurement.estimate_error_ratio).toFixed(1)}×`,
};
}

function rememberWorkloadActions(actions, runs) {
const currentBaseline = {};
(runs.items || []).forEach((run) => {
if (run.phase === "baseline" && currentBaseline[run.scenario] === undefined) currentBaseline[run.scenario] = run.id;
});
(actions.items || []).forEach((action) => {
const source = action.validation?.proposal_source;
if (source?.type !== "workload_run" || !source.scenario) return;
if (state.baselineRuns[source.scenario] === source.run_id || currentBaseline[source.scenario] === source.run_id) {
if (state.baselineRuns[source.scenario] === undefined) state.baselineRuns[source.scenario] = source.run_id;
if (state.actionsByScenario[source.scenario] === undefined || Number(action.id) > Number(state.actionsByScenario[source.scenario])) state.actionsByScenario[source.scenario] = action.id;
}
});
}

async function loadWorkloadRuns() {
const [data, actions] = await Promise.all([api("/api/workload-runs?limit=50"), api("/api/actions?limit=200")]);
rememberActionRecords(actions.items);
renderGuidedDemo();
rememberWorkloadActions(actions, data);
const latestRuns = {};
data.items.forEach((row) => {
if (row.status !== "completed" || !["baseline", "after"].includes(row.phase)) return;
const key = row.scenario + ":" + row.phase;
if (!latestRuns[key] || Number(row.id) > Number(latestRuns[key].id)) latestRuns[key] = row;
});
Object.keys(latestRuns).forEach((key) => {
const row = latestRuns[key];
const scenarioName = row.scenario;
if (row.phase === "baseline" && state.baselineRuns[scenarioName] === undefined) {
state.baselineRuns[scenarioName] = row.id;
state.baselineResults[scenarioName] = rowToRunResult(row);
}
if (row.phase === "after" && state.afterRuns[scenarioName] === undefined) {
const baseline = latestRuns[scenarioName + ":baseline"];
if (!baseline || Number(row.id) > Number(baseline.id)) {
state.afterRuns[scenarioName] = row.id;
state.afterResults[scenarioName] = rowToRunResult(row);
}
}
});

function renderRagStatus(data) {
const database = data.database || {};
const types = database.source_types || [];
const selector = $("ragSourceType");
const selected = selector.value;
selector.innerHTML = '<option value="">All indexed sources</option>' + types.map((row) => '<option value="' + esc(row.source_type) + '">' + esc(row.source_type) + ' (' + fmt(row.documents, 0) + ')</option>').join("");
if ([...selector.options].some((option) => option.value === selected)) selector.value = selected;
const indexer = data.indexer || {};
const vector = data.vector || {};
const evaluation = data.evaluation?.report?.summary;
const staticManifest = data.manifest?.rag?.static_manifest || {};
const evaluationText = data.evaluation?.available && evaluation
? `Fixed retrieval evaluation: ${fmt(evaluation.pass_count, 0)}/${fmt(evaluation.question_count, 0)} passed · Recall@1 ${(Number(evaluation.source_recall_at_1 || 0) * 100).toFixed(1)}% · Recall@k ${(Number(evaluation.source_recall_at_k || 0) * 100).toFixed(1)}% · MRR ${Number(evaluation.mean_reciprocal_rank || 0).toFixed(2)}.`
: `Fixed retrieval evaluation unavailable: ${esc(data.evaluation?.reason || "run scripts/evaluate_rag.py")}`;
$("ragStatus").innerHTML = '<div class="rag-status-grid">' +
'<div class="rag-status-item"><span>Documents</span><strong>' + fmt(database.documents, 0) + '</strong></div>' +
'<div class="rag-status-item"><span>Chunks</span><strong>' + fmt(database.chunks, 0) + '</strong></div>' +
'<div class="rag-status-item"><span>Last sync</span><strong>' + esc(date(indexer.last_sync_at || database.latest_update)) + '</strong></div>' +
'<div class="rag-status-item"><span>Background sync</span><strong>' + (indexer.running ? 'running' : 'stopped') + '</strong></div></div>' +
'<p class="tiny muted">' + (indexer.last_error ? 'Last sync error: ' + esc(indexer.last_error) : 'Static sources are hash-checked; runtime workload, action, feedback, incident, knowledge, and telemetry records are periodically mirrored.') + '</p>' +
'<p class="tiny muted">' + (staticManifest.available ? 'Static corpus: ' + fmt(staticManifest.record_count, 0) + ' records · SHA-256 ' + esc(String(staticManifest.sha256 || '').slice(0, 16)) + '…' : 'Static corpus manifest unavailable') + ' · Embeddings: ' + fmt(vector.embeddings, 0) + ' × ' + fmt(vector.dimensions, 0) + ' (' + esc(vector.model_name || 'not configured') + ')</p>' +
'<p class="tiny muted">' + evaluationText + '</p>';
}

function renderRagResults(data) {
const items = data.results || [];
$("ragResults").innerHTML = items.length

' · ' + esc(locatorText(item.locator)) : '') + '</div></article>').join("")
: '<div class="empty">No passages retrieved yet.</div>';
}

function renderRagAnswer(data) {
