function chartNumber(value) {
if (value === null || value === undefined || value === "") return null;
const number = Number(value);
return Number.isFinite(number) ? number : null;
}

function medianNumbers(values) {
const sorted = values.filter((value) => value !== null && value !== undefined).slice().sort((a, b) => a - b);
if (!sorted.length) return null;
const middle = Math.floor(sorted.length / 2);
return sorted.length % 2 ? sorted[middle] : (sorted[middle - 1] + sorted[middle]) / 2;
}

function stableChartTrend(values, radius = 2) {
return values.map((value, index) => {
if (value === null) return null;
const window = values.slice(Math.max(0, index - radius), Math.min(values.length, index + radius + 1));
return medianNumbers(window) ?? value;
});
}

function detectChartEvents(values, options = {}) {
const events = [];
const historySize = options.historySize || 5;
const absoluteThreshold = options.absoluteThreshold ?? 2;
const relativeThreshold = options.relativeThreshold ?? .75;
values.forEach((value, index) => {
if (value === null) return;
const history = values.slice(Math.max(0, index - historySize), index).filter((item) => item !== null);
if (history.length < 3) return;
const baseline = medianNumbers(history);
if (baseline === null) return;
const deviations = history.map((item) => Math.abs(item - baseline));
const mad = medianNumbers(deviations) || 0;
const threshold = Math.max(absoluteThreshold, Math.abs(baseline) * relativeThreshold, mad * 4);
const delta = value - baseline;
const rise = delta >= threshold && value >= baseline * 1.5;
const drop = baseline > 0 && value <= baseline * .35;
const zeroToPositive = baseline === 0 && value >= absoluteThreshold;
if (rise || drop || zeroToPositive) {
events.push({ index, value, baseline, delta, direction: delta >= 0 ? "up" : "down" });
}
});
return events;
}

function drawSeriesPath(ctx, values, pointX, pointY, color, lineWidth, alpha = 1) {
ctx.save();
ctx.globalAlpha = alpha;
ctx.strokeStyle = color;
ctx.lineWidth = lineWidth;
ctx.lineJoin = "round";
ctx.lineCap = "round";
ctx.beginPath();
let drawing = false;
values.forEach((value, index) => {
if (value === null) { drawing = false; return; }
const point = { x: pointX(index), y: pointY(value) };
if (!drawing) { ctx.moveTo(point.x, point.y); drawing = true; } else ctx.lineTo(point.x, point.y);
});
ctx.stroke();
ctx.restore();
}

const max = fixedMax === null

const pointX = (index) => pad.left + (values.length <= 1 ? chartW / 2 : chartW * index / (values.length - 1));
const pointY = (value) => pad.top + chartH * (1 - Math.min(1, Math.max(0, value) / max));
drawSeriesPath(ctx, values, pointX, pointY, color, options.rawLineWidth || 1.1, options.rawAlpha ?? .28);
drawSeriesPath(ctx, trend, pointX, pointY, color, options.trendLineWidth || 2.5, 1);
if (options.showRawPoints) {
ctx.save();
ctx.fillStyle = color;
ctx.globalAlpha = .55;
values.forEach((value, index) => {
if (value === null) return;
ctx.beginPath();
ctx.arc(pointX(index), pointY(value), 1.8, 0, Math.PI * 2);
ctx.fill();
});
ctx.restore();
}
if (options.showEvents && events.length) {
ctx.save();
ctx.strokeStyle = "#d65a3a";
ctx.fillStyle = "#d65a3a";
ctx.globalAlpha = .8;
ctx.setLineDash([3, 4]);
events.forEach((event) => {
const x = pointX(event.index);
ctx.beginPath();
ctx.moveTo(x, pad.top);
ctx.lineTo(x, height - pad.bottom);
ctx.stroke();
ctx.setLineDash([]);
ctx.beginPath();
ctx.arc(x, pointY(event.value), 3.3, 0, Math.PI * 2);
ctx.fill();
ctx.setLineDash([3, 4]);
});
ctx.restore();
// Draw a small exact-value label for the most recent events. If a
// burst is much larger than the stable axis, its point is clipped at
// the top but the legend and label still expose the raw value.
ctx.save();
ctx.font = "10px system-ui, sans-serif";
ctx.textBaseline = "middle";
events.slice(-3).forEach((event, labelIndex) => {
const arrow = event.direction === "up" ? "↑" : "↓";
const label = arrow + fmt(event.value, 2) + unit;
const textWidth = ctx.measureText(label).width;
const x = Math.min(Math.max(pointX(event.index) - textWidth / 2, pad.left), width - pad.right - textWidth);
const y = pad.top + 9 + (labelIndex % 2) * 13;
ctx.fillStyle = "rgba(255,255,255,.92)";
ctx.fillRect(x - 3, y - 7, textWidth + 6, 14);
ctx.fillStyle = "#b14b35";
ctx.fillText(label, x, y);
});
ctx.restore();
}
ctx.fillStyle = "#8b96a6";
ctx.textAlign = "left";
const first = items[0]?.collected_at;
const last = items[items.length - 1]?.collected_at;
ctx.fillText(chartTimeLabel(first), pad.left, height - 7);
ctx.textAlign = "right";
ctx.fillText(chartTimeLabel(last), width - pad.right, height - 7);
ctx.textAlign = "left";
renderChartLegend(options.legendId, items, unit, events, observedMax, max, { values });
}

function drawOverviewCharts(items) {
const transactionRate = (item) => {
const rates = compactRates(item);
const commit = chartNumber(rates.xact_commit), rollback = chartNumber(rates.xact_rollback);
return commit === null && rollback === null ? null : (commit || 0) + (rollback || 0);
};
const queryP95 = (item) => {
const latency = compactLatency(item);
return latency.available ? latency.p95_ms : null;
};
const connections = (item) => compactDatabase(item).numbackends;
const cacheRatio = (item) => {
const database = compactDatabase(item);
const hit = chartNumber(database.blks_hit), read = chartNumber(database.blks_read);
return hit === null || read === null || hit + read <= 0 ? null : (hit / (hit + read)) * 100;
};
const tpsOptions = {
legendId: "tpsChartLegend",
preserveEvents: true,
showEvents: true,
showRawPoints: true,
absoluteThreshold: 2,
};
drawLineChart("tpsChart", items, transactionRate, "#3370ff", "/秒", null, tpsOptions);
drawLineChart("workloadTpsChart", items, transactionRate, "#3370ff", "/秒", null, { ...tpsOptions, legendId: "workloadTpsLegend" });
["latencyChart", "workloadLatencyChart"].forEach((id) => drawLineChart(id, items, queryP95, "#e86a2a", "ms"));
["connectionsChart", "workloadConnectionsChart"].forEach((id) => drawLineChart(id, items, connections, "#7b61c9", "个"));
["cacheChart", "workloadCacheChart"].forEach((id) => drawLineChart(id, items, cacheRatio, "#0f9d68", "%", 100));
}

function drawChart(items) { drawOverviewCharts(items); }

if (!detail || !report) return;
const run = report.run || {};
const diagnosis = report.diagnosis || {};
const llm = diagnosis.llm_analysis;
const causes = (diagnosis.root_causes || []).map((cause) =>
'<span class="badge info">' + esc(anomalyTitle(cause)) + "</span>"
).join(" ") || '<span class="muted">尚未确定根因</span>';
const actions = (diagnosis.actions || []).slice(0, 4).map((action) =>
'<div class="report-action"><code>' + esc(localActionType(action.action_type)) +
'</code><span>' + esc(pretty(action.target || action)) + "</span></div>"
).join("") || '<span class="muted">当前证据没有生成类型化调优建议。</span>';
const evidence = (diagnosis.evidence || []).slice(0, 4).map((item) =>
'<div class="report-evidence"><strong>' + esc(item.source || "PostgreSQL 证据") +
'</strong><span>' + esc(item.interpretation || pretty(item.observed || {})) + "</span></div>"
).join("");
const llmBlock = llm
? '<div class="report-llm"><div class="eyebrow">大模型分析结果</div><pre>' +
esc(llm.text || "") + '</pre><p class="tiny muted">模型：' + esc(llm.model || "—") +
" · 引用：" + esc((llm.evidence_ids || []).join(", ")) + " · 仅作证据解释</p></div>"
: '<div class="report-llm report-llm-empty"><div class="eyebrow">大模型分析结果</div><p>本次没有返回外部模型分析；规则诊断和 PostgreSQL 原始证据仍然保留。</p></div>';
detail.classList.remove("hidden");
detail.innerHTML =
'<div class="report-detail-head"><div><span class="eyebrow">诊断报告 · 真实运行 #' +
esc(run.run_id ?? run.id ?? "—") + "</span><h3>" + esc(anomalyTitle(run.scenario || report.incident?.kind)) +
'</h3></div><button class="text-button" data-report-close>关闭</button></div>' +
(report.error ? '<div class="notice">' + esc(report.error) + "</div>" : "") +
'<div class="report-causes"><span class="diagnosis-label">根因</span>' + causes + "</div>" +
'<div class="report-columns"><div><span class="diagnosis-label">类型化建议</span><div class="report-actions">' +
actions + '</div></div><div><span class="diagnosis-label">证据摘要</span><div class="report-evidence-list">' +
(evidence || '<span class="muted">没有可折叠的诊断证据。</span>') + "</div></div></div>" +
llmBlock +
'<div class="report-detail-actions"><button class="button small secondary" data-report-actions>查看调优操作</button><span class="tiny muted">是否应用由审核页控制</span></div>';
detail.querySelector("[data-report-close]").onclick = () => detail.classList.add("hidden");
detail.querySelector("[data-report-actions]").onclick = () => setView("actions");
}

function waitForWorkload(ms) {
return new Promise((resolve) => window.setTimeout(resolve, ms));
}

async function createWorkloadReport(result) {
state.workloadFlow.runId = result.run_id;
renderWorkloadFlowStatus("运行 #" + result.run_id + " 已完成，正在生成诊断报告…", "busy");
try {
const data = await api("/api/workload-runs/" + result.run_id + "/diagnose", { method: "POST" });
const report = { run: result, diagnosis: data.diagnosis || data };
state.diagnoses[result.run_id] = data;
upsertWorkloadReport(report);
renderWorkloadReports();
renderWorkloadFlowStatus("报告已生成 · 点击右侧链接查看大模型分析", "ok");
toast("诊断报告已生成，规则证据与大模型分析均已保存");
return report;
} catch (e) {
const report = { run: result, error: e.message };
upsertWorkloadReport(report);
renderWorkloadReports();
renderWorkloadFlowStatus("测量完成，但报告生成失败：" + e.message, "bad");
toast("测量完成，但诊断报告生成失败：" + e.message, true);
return report;
}
}

async function triggerLockAnomaly() {
const startedAt = Date.now();
let started;
let incident = null;
let diagnosis = null;
try {
started = await api("/api/workloads/lock_contention/start", { method: "POST" });
state.lockRunId = Number(started.run_id);
renderWorkloadFlowStatus("锁竞争已触发，等待 pg_locks 证据…", "busy");
for (let attempt = 0; attempt < 11 && !incident; attempt += 1) {
const data = await optionalApi("/api/incidents?kind=lock_wait&status=active&limit=30");
incident = (data.items || []).find((item) => {
const seen = Date.parse(item.last_seen || item.first_seen || "");
return !Number.isFinite(seen) || seen >= startedAt - 2000;
});
if (!incident) await waitForWorkload(1000);
}
if (incident) {
renderWorkloadFlowStatus("锁等待已被采集，正在生成诊断报告…", "busy");
const data = await api("/api/incidents/" + incident.id + "/diagnose", { method: "POST" });
diagnosis = data.diagnosis || data;
}
} catch (e) {
diagnosis = null;
if (!started) throw e;
} finally {
if (state.lockRunId) {
try {
await api("/api/workloads/lock_contention/" + state.lockRunId + "/stop", { method: "POST" });
} catch (e) {
toast("锁竞争停止失败：" + e.message, true);
}
state.lockRunId = null;
}
await Promise.all([refreshDashboard(), loadIncidents(), loadWorkloadRuns()]);
}
const run = {
run_id: started?.run_id,
scenario: "lock_contention",
phase: "contention",
status: "completed",
started_at: started?.started_at,
message: incident ? "已采集锁等待事件" : "锁等待已真实触发并释放",
};
const report = incident && diagnosis
? { run: run, incident: incident, diagnosis: diagnosis }
: {
run: run,
incident: incident,
error: "锁等待已真实触发并释放，但采集器尚未在本窗口生成 lock_wait 事件。",
};
upsertWorkloadReport(report);
renderWorkloadReports();
renderWorkloadFlowStatus(report.error ? report.error : "报告已生成 · 点击右侧链接查看大模型分析", report.error ? "bad" : "ok");
if (report.error) toast(report.error, true);
else toast("锁竞争诊断报告已生成");
return report;
}

async function triggerAnomaly(scenarioName) {
if (state.workloadFlow.busy) return toast("当前已有异常测量在运行，请等待报告完成", true);
const scenario = state.workloadDefinitions.find((item) => item.name === scenarioName);
if (!scenario) return toast("异常场景仍在加载，请稍候", true);
state.workloadFlow = { busy: true, scenario: scenarioName, runId: null, message: "", tone: "busy" };
renderWorkloadFlowStatus("正在启动 " + anomalyTitle(scenarioName) + "…", "busy");
try {
if (scenarioName === "lock_contention") {
await triggerLockAnomaly();
} else {
const result = await runWorkloadPhase(
scenarioName,
"baseline",
readWorkloadParameters(scenarioName),
{ stayOnWorkloads: true }
);
await createWorkloadReport(result);
}
} catch (e) {
renderWorkloadFlowStatus("异常触发失败：" + e.message, "bad");
toast(e.message, true);
} finally {
state.workloadFlow.busy = false;
renderWorkloadFlowStatus(state.workloadFlow.message || "等待异常触发", state.workloadFlow.tone || "");
renderNormalLoadStatus(state.normalLoad.active || {});
}
}

function readWorkloadParameters(scenarioName) {
const definition = state.workloadDefinitions.find((scenario) => scenario.name === scenarioName);
const parameters = { ...(definition?.parameters || {}) };
const card = document.querySelector('.workload-card[data-scenario="' + scenarioName + '"]');
if (card) {
card.querySelectorAll("[data-workload-param]").forEach((input) => {
parameters[input.dataset.workloadParam] = input.value.trim();
});
}
return parameters;
}

function rememberRun(result) {
state.lastRun = result;
state.guidedCurrent[result.scenario] = true;
if (result.phase === "baseline") {
state.baselineRuns[result.scenario] = result.run_id;
state.baselineResults[result.scenario] = result;
state.actionsByScenario[result.scenario] = undefined;
state.actionRecords[result.scenario] = undefined;
state.feedbackByScenario[result.scenario] = undefined;
state.afterRuns[result.scenario] = undefined;
state.afterResults[result.scenario] = undefined;
}
if (result.phase === "after") {
state.afterRuns[result.scenario] = result.run_id;
state.afterResults[result.scenario] = result;
}
}

function clearRunnerTimers() {
if (state.runner.pollTimer) clearInterval(state.runner.pollTimer);
if (state.runner.elapsedTimer) clearInterval(state.runner.elapsedTimer);
state.runner.pollTimer = null;
state.runner.elapsedTimer = null;
state.runner.pollInFlight = false;
}

function runnerElapsedSeconds() {
if (!state.runner.startedAt) return 0;
return Math.max(0, (Date.now() - state.runner.startedAt) / 1000);
}

function startRunnerClock() {
if (state.runner.elapsedTimer) clearInterval(state.runner.elapsedTimer);
const update = () => {
const node = $("workloadRunnerElapsed");
if (node) node.textContent = `${runnerElapsedSeconds().toFixed(1)} 秒`;
};
update();
state.runner.elapsedTimer = setInterval(update, 100);
}

function runnerStageText(stage, status) {
if (status === "completed") return "测量完成";
if (status === "failed") return "测量失败";
return ({ queued: "已排队，等待后台线程", starting: "正在连接 PostgreSQL", warming_up: "正在预热真实查询", measuring: "正在持续计时样本" }[stage] || "正在运行真实 PostgreSQL 测量");
}

function renderWorkloadRunner(snapshot = {}) {
const runner = state.runner;
const row = snapshot && Object.keys(snapshot).length ? snapshot : (runner.snapshot || {});
const observations = row.observations || {};
const progress = observations.progress || {};
const status = row.status || (runner.running ? "running" : "pending");
const scenario = state.workloadDefinitions.find((item) => item.name === (row.scenario || runner.scenario)) || scenarioText[row.scenario || runner.scenario] || {};
const scenarioName = row.scenario || runner.scenario || "";
const total = Math.max(0, Number(progress.total_samples || (Number(row.repetitions || runner.repetitions || 0) + Number(row.warmup_repetitions || runner.warmupRepetitions || 0))));
const completed = status === "completed" ? total : Math.min(total, Math.max(0, Number(progress.completed_samples || 0)));
const percent = total ? Math.round((completed / total) * 100) : 0;
const stage = runnerStageText(progress.stage, status);
const phaseText = row.phase === "after" ? "调优后版本" : row.phase === "baseline" ? "基线版本" : "工作负载";
const timingSamples = Array.isArray(progress.timings_ms) && progress.timings_ms.length ? progress.timings_ms : (Array.isArray(observations.timings_ms) ? observations.timings_ms : []);
const warmups = Number(progress.warmup_samples || 0);
const lastSample = progress.last_sample_ms;
const titleNode = $("workloadRunnerTitle");

titleNode.textContent = scenarioName ? `${scenario.title || scenarioText[scenarioName]?.title || scenarioName} · ${phaseText}` : "等待启动真实工作负载";

}

async function pollWorkloadRun(started) {
const runId = Number(started.run_id);
return new Promise((resolve, reject) => {
let settled = false;
const finish = (error, result) => {
if (settled) return;
settled = true;
clearRunnerTimers();
state.runner.running = false;
if (error) reject(error); else resolve(result);
};
const check = async () => {
if (settled || state.runner.pollInFlight) return;
state.runner.pollInFlight = true;
try {
const snapshot = await api(`/api/workload-runs/${runId}`);
if (state.runner.runId !== runId) return finish(new Error("当前实时测量已被另一个运行替换"));
state.runner.snapshot = snapshot;
const activity = await optionalApi("/api/activity?limit=12", { items: [] });
state.runner.activity = activity.items || [];
renderWorkloadRunner(snapshot);
if (state.workloadFlow.busy && state.workloadFlow.scenario === state.runner.scenario) {
const progress = snapshot.observations?.progress || {};
const completed = Number(progress.completed_samples || 0);
const total = Number(progress.total_samples || 0);
renderWorkloadFlowStatus(
anomalyTitle(state.runner.scenario) + " · 已完成 " + completed + (total ? " / " + total : "") + " 个真实样本",
"busy"
);
}
if (snapshot.status === "completed") {
const result = rowToRunResult(snapshot);
state.runner.completedResult = result;
return finish(null, result);
}
if (snapshot.status === "failed") return finish(new Error(snapshot.error || "工作负载运行失败"));
} catch (error) {
state.runner.error = error.message;
renderWorkloadRunner(state.runner.snapshot || started);
return finish(error);
} finally {
state.runner.pollInFlight = false;
}
};
state.runner.pollTimer = setInterval(check, 400);
check();
});
}

async function runWorkloadPhase(scenarioName, phase, parameters = null, options = {}) {
if (phase === "after" && !state.baselineRuns[scenarioName]) {
throw new Error("请先完成该场景的基线测量");
}
clearRunnerTimers();
state.runner = { runId: null, scenario: scenarioName, phase, repetitions: workloadRepetitions(), warmupRepetitions: workloadWarmups(), startedAt: Date.now(), running: true, snapshot: null, completedResult: null, activity: [], pollTimer: null, elapsedTimer: null, pollInFlight: false, error: null };
state.guidedScenario = scenarioName;
state.guidedCurrent[scenarioName] = true;
if (!options.stayOnWorkloads) setView("workload-runner");
startRunnerClock();
renderWorkloadRunner({ scenario: scenarioName, phase, status: "running", repetitions: state.runner.repetitions, warmup_repetitions: state.runner.warmupRepetitions, observations: { progress: { stage: "queued", total_samples: state.runner.repetitions + state.runner.warmupRepetitions } } });
try {
const started = await api(`/api/workloads/${scenarioName}/runs/start`, {
method: "POST",
body: JSON.stringify({ phase, repetitions: state.runner.repetitions, warmup_repetitions: state.runner.warmupRepetitions, parameters: parameters || readWorkloadParameters(scenarioName) }),
});
state.runner.runId = Number(started.run_id);
if (state.workloadFlow.busy && state.workloadFlow.scenario === scenarioName) state.workloadFlow.runId = state.runner.runId;
state.runner.snapshot = started;
state.runner.repetitions = Number(started.repetitions || state.runner.repetitions);
state.runner.warmupRepetitions = Number(started.warmup_repetitions || state.runner.warmupRepetitions);
renderWorkloadRunner(started);
const result = await pollWorkloadRun(started);
rememberRun(result);
state.lastRun = result;
state.runner.completedResult = result;
state.runner.snapshot = { ...state.runner.snapshot, ...result, status: "completed", observations: result };
renderWorkloadRunner(state.runner.snapshot);
if (!options.stayOnWorkloads) toast(`${phase === "baseline" ? "基线" : "调优后"}测量已完成，数据来自 PostgreSQL；请进入下一步查看证据`);
await Promise.all([loadWorkloadRuns(), refreshDashboard()]);
renderGuidedDemo();
return result;
} catch (error) {
state.runner.running = false;
state.runner.error = error.message;
renderWorkloadRunner(state.runner.snapshot || {});
throw error;
}
}

function continueFromRunner() {
const result = state.runner.completedResult;
if (!result) return;
setView("workloads");
state.lastRun = result;
showWorkloadResult(result);
renderGuidedDemo();
const nextStep = result.phase === "after" ? "反馈" : "操作建议";
toast(`已回到工作负载流程，下一步是${nextStep}`);
}

function rememberActionRecords(items) {
(items || []).forEach((row) => {
const source = row.validation?.proposal_source;
if (source?.type !== "workload_run" || !source.scenario) return;
const currentBaseline = state.baselineRuns[source.scenario];
if (currentBaseline !== undefined && Number(currentBaseline) !== Number(source.run_id)) return;
const current = state.actionRecords[source.scenario];
if (!current || Number(row.id) > Number(current.id)) {
state.actionRecords[source.scenario] = row;
}
if (row.validation?.latest_feedback) {
state.feedbackByScenario[source.scenario] = row.validation.latest_feedback;
}
});
}

}

function updateActionRecord(action) {
const source = action?.validation?.proposal_source;
if (!source?.scenario) return;
state.actionsByScenario[source.scenario] = action.id;
state.actionRecords[source.scenario] = action;
}

async function runGuidedStep(step) {
if (state.guidedBusy) return;
const scenario = state.guidedScenario;
state.guidedBusy = true;
renderGuidedDemo();
try {
if (step === "baseline") {
await runWorkloadPhase(scenario, "baseline");
} else if (step === "propose") {
const action = await api("/api/workload-runs/" + state.baselineRuns[scenario] + "/propose", { method: "POST" });
updateActionRecord(action);
state.lastRun = state.baselineResults[scenario];
showWorkloadResult(state.lastRun);
await loadActions();
toast("操作建议 #" + action.id + " 已生成，等待审核");
} else if (step === "apply") {
const action = await api("/api/actions/" + state.actionsByScenario[scenario] + "/apply", { method: "POST" });
updateActionRecord(action);
await loadActions();
toast("已审核的调优操作已应用，并在 PostgreSQL 中完成验证");
} else if (step === "after") {
await runWorkloadPhase(scenario, "after");
} else if (step === "feedback") {
const data = await api("/api/actions/" + state.actionsByScenario[scenario] + "/feedback", {
method: "POST",
body: JSON.stringify({
baseline_run_id: Number(state.baselineRuns[scenario]),
after_run_id: Number(state.afterRuns[scenario]),
}),
});
state.feedbackByScenario[scenario] = data.feedback?.outcome || {};
updateActionRecord(data.action);
await loadActions();
await loadKnowledge();
showWorkloadResult(state.afterResults[scenario]);
toast("实测结果已记录：" + (data.feedback?.outcome?.observed_effect || "完成"));
} else if (step === "rollback") {
const action = await api("/api/actions/" + state.actionsByScenario[scenario] + "/rollback", { method: "POST" });
updateActionRecord(action);
await loadActions();
await refreshDashboard();
toast("调优操作已回滚，并已验证 PostgreSQL 状态");
}
} catch (e) {
toast(e.message, true);
} finally {
state.guidedBusy = false;
renderGuidedDemo();
}
}

async function applyAutomaticResolution(scenario, baselineRunId) {
const data = await api(`/api/workload-runs/${baselineRunId}/auto-resolve`, {
method: "POST",
body: JSON.stringify({ rollback_after_success: Boolean($("guidedAutoRollback")?.checked) }),
});
const action = data.action || {};
const after = data.after || {};
const feedback = data.feedback?.feedback?.outcome || {};
state.guidedCurrent[scenario] = true;
state.baselineRuns[scenario] = Number(data.baseline?.id || baselineRunId);
state.baselineResults[scenario] = { ...(data.baseline || {}), run_id: state.baselineRuns[scenario], scenario, phase: "baseline", observations: data.baseline?.observations || {} };
state.actionsByScenario[scenario] = action.id;
state.actionRecords[scenario] = action;
state.afterRuns[scenario] = Number(after.run_id);
state.afterResults[scenario] = after;
state.feedbackByScenario[scenario] = feedback;
state.lastRun = after;
showWorkloadResult(after);
await Promise.all([loadWorkloadRuns(), loadActions(), loadKnowledge(), refreshDashboard()]);
toast(`自动闭环完成：${data.resolution?.effective ? "调优有效" : "调优无效"}${data.resolution?.rollback_performed ? "；状态已恢复" : "；操作仍已部署"}`);
return data;
}

return value;
});

function comparisonEvidence(result) {

const baseline = baselineResult?.observations || baselineResult;
const after = result.observations || result;
if (!baseline || result.phase !== "after") return "";
const baseMedian = Number(baseline.median_query_time_ms);
const afterMedian = Number(after?.median_query_time_ms);

const afterP95 = Number(after?.p95_query_time_ms);
const improvement = (before, current) => before > 0 && Number.isFinite(before) && Number.isFinite(current) ? ((before - current) / before) * 100 : null;
const medianImprovement = improvement(baseMedian, afterMedian);
const p95Improvement = improvement(baseP95, afterP95);
const recorded = state.feedbackByScenario[result.scenario];
const status = recorded
? `<span class="badge ${recorded.effective ? "applied" : "warning"}">${esc(localStatus(recorded.observed_effect || "recorded"))}</span>`
: `<span class="badge info">等待服务器反馈</span>`;
return `<div class="comparison-panel">
<div class="comparison-header"><div><span class="eyebrow">实测对比</span><h3>基线 → 调优后</h3></div>${status}</div>
<div class="comparison-grid">
${workloadMetric("中位耗时", `${fmt(baseMedian)} → ${fmt(afterMedian)} 毫秒`, `实测改善 ${formatPercent(medianImprovement)}`, medianImprovement !== null && medianImprovement > 0 ? "good" : "bad")}
${workloadMetric("P95 耗时", `${fmt(baseP95)} → ${fmt(afterP95)} 毫秒`, `实测改善 ${formatPercent(p95Improvement)}`, p95Improvement !== null && p95Improvement > 0 ? "good" : "bad")}
</div>
<p class="tiny muted">此处使用两次返回的实测结果。服务器只有在通过执行计划、结果、参数和重复次数校验后，才会将反馈记录为 <code>effective</code>。</p>
</div>`;
}

function planSignals(observations) {
const summary = observations?.plan_summary || {};
const nodes = observations?.plan?.nodes || [];
const hashes = nodes
.filter((node) => node["Node Type"] === "Hash")
.map((node) => node["Hash Batches"] ?? node.Batches)
.filter((value) => value !== undefined);
const sorts = summary.sort_methods || nodes
.filter((node) => node["Sort Method"])
.map((node) => node["Sort Method"]);
const sortSpaceTypes = summary.sort_space_types || nodes
.filter((node) => node["Sort Space Type"])
.map((node) => node["Sort Space Type"]);
const aggregates = summary.aggregate_strategies || nodes
.filter((node) => ["Aggregate", "HashAggregate"].includes(node["Node Type"]) && node.Strategy)
.map((node) => node.Strategy);
const temporary = Boolean(summary.has_temp_blocks) || nodes.some((node) => (
Number(node["Temp Read Blocks"] || 0) > 0 || Number(node["Temp Written Blocks"] || 0) > 0
));
return {
root: summary.root_node || observations?.plan?.root?.["Node Type"] || "未观测",
scans: (summary.scan_nodes || []).join(" → ") || "未观测",
joins: (summary.join_nodes || []).join(" → ") || "未观测",
hashBatches: hashes.length ? hashes.join(", ") : "未观测",
sortMethods: sorts.length ? sorts.join(", ") : "未观测",
sortSpaceTypes: sortSpaceTypes.length ? sortSpaceTypes.join(", ") : "未观测",
aggregates: aggregates.length ? aggregates.join(", ") : "未观测",
temporary: temporary ? "已观测" : "未观测",
workers: Math.max(0, ...nodes.map((node) => Number(node["Workers Launched"] || 0))),
plannedWorkers: Math.max(0, ...nodes.map((node) => Number(node["Workers Planned"] || 0))),
indexes: nodes.map((node) => node["Index Name"]).filter(Boolean).join(", ") || "未观测",
estimateError: observations?.statistics_measurement?.estimate_error_ratio === undefined ? "未观测" : `${Number(observations.statistics_measurement.estimate_error_ratio).toFixed(1)}×`,
};
}

}

function rememberWorkloadActions(actions, runs) {
const currentBaseline = {};
(runs.items || []).forEach((run) => {
if (run.phase === "baseline" && currentBaseline[run.scenario] === undefined) currentBaseline[run.scenario] = run.id;
});
(actions.items || []).forEach((action) => {
const source = action.validation?.proposal_source;
if (source?.type !== "workload_run" || !source.scenario) return;
if (state.baselineRuns[source.scenario] === source.run_id || currentBaseline[source.scenario] === source.run_id) {
if (state.baselineRuns[source.scenario] === undefined) state.baselineRuns[source.scenario] = source.run_id;
if (state.actionsByScenario[source.scenario] === undefined || Number(action.id) > Number(state.actionsByScenario[source.scenario])) state.actionsByScenario[source.scenario] = action.id;
}
});
}

try { const result = await api("/api/rag/vector/sync?limit=10000", { method: "POST" }); toast(`向量索引已同步：新增 ${fmt(result.sync?.created, 0)} 条，更新 ${fmt(result.sync?.updated, 0)} 条`); await loadRag(); }
catch (e) { toast(e.message, true); } finally { button.disabled = false; }
}

async function exportEvidence() {
const button = $("exportEvidenceButton");
if (button) button.disabled = true;
try {
const data = await api("/api/export?limit=1000");
const blob = new Blob([JSON.stringify(data, null, 2)], { type: "application/json" });
const url = URL.createObjectURL(blob);
const anchor = document.createElement("a");
anchor.href = url;
anchor.download = "keeninsight-postgresql-evidence-" + new Date().toISOString().replace(/[:.]/g, "-") + ".json";
document.body.appendChild(anchor);
anchor.click();
anchor.remove();
URL.revokeObjectURL(url);
toast("证据已导出，包含运行、执行计划、调优操作、反馈和审计记录");
} catch (e) {
toast(e.message, true);
} finally {
if (button) button.disabled = false;
}
}

async function cleanDemoState() {
if (!window.confirm("要回滚 KeenInsight 演示操作并释放演示锁会话吗？历史证据会保留。")) return;
const button = $("cleanDemoButton");
if (button) button.disabled = true;
try {
const result = await api("/api/demo/clean", { method: "POST" });
state.baselineRuns = {};
state.baselineResults = {};
state.actionsByScenario = {};
state.actionRecords = {};
state.feedbackByScenario = {};
state.afterRuns = {};
state.afterResults = {};
state.diagnoses = {};
state.guidedCurrent = {};
if (result.rollback_errors?.length) {
toast("演示清理完成，但有 " + result.rollback_errors.length + " 个回滚错误；请检查调优操作表", true);
} else {
toast("演示状态已清理；历史证据已保留");
}
$("workloadResultCard").classList.add("hidden");
await loadWorkloads();
await refreshDashboard();
} catch (e) {
toast(e.message, true);
} finally {
if (button) button.disabled = false;
}
}

function bind() {
document.querySelectorAll("[data-view]").forEach((node) => node.onclick = () => setView(node.dataset.view)); document.querySelectorAll("[data-view-link]").forEach((node) => node.onclick = () => setView(node.dataset.viewLink)); $("refreshButton").onclick = refreshDashboard; $("autoRefresh").onchange = (e) => { state.autoRefresh = e.target.checked; if (state.autoRefresh) schedule(); else clearTimeout(state.timer); }; $("incidentRefresh").onclick = loadIncidents; $("incidentStatusFilter").onchange = loadIncidents; $("closeIncidentDetail").onclick = () => $("incidentDetailCard").classList.add("hidden"); $("queryRefresh").onclick = loadQueries; $("explainButton").onclick = explain; $("bootstrapButton").onclick = async () => { if (!confirm("要创建或刷新命名空间演示数据吗？")) return; try { await api("/api/demo/bootstrap", { method: "POST" }); toast("演示关系表和统计信息已准备就绪"); await loadWorkloads(); } catch (e) { toast(e.message, true); } }; $("workloadRefresh").onclick = loadWorkloads; $("workloadRunnerBackButton").onclick = () => setView("workloads"); $("workloadRunnerBackToWorkload").onclick = () => setView("workloads"); $("workloadRunnerNextButton").onclick = continueFromRunner; $("actionRefresh").onclick = loadActions; $("knowledgeRefresh").onclick = loadKnowledge; $("settingsRefresh").onclick = loadSettings; $("rulesRefresh").onclick = loadRules; $("settingsSearch").onkeydown = (e) => { if (e.key === "Enter") loadSettings(); }; $("auditRefresh").onclick = loadAudit; $("observabilityRefresh").onclick = loadObservability; $("ebpfEventsRefresh").onclick = loadObservability; $("ebpfStartButton").onclick = startEbpf; $("ebpfStopButton").onclick = stopEbpf; $("ebpfAdaptButton").onclick = adaptEbpf; if ($("guidedAutoResolveButton")) $("guidedAutoResolveButton").onclick = autoResolveGuided; if ($("guidedPressureAutoButton")) $("guidedPressureAutoButton").onclick = runPressureAndAutoRepair; $("analysisRefresh").onclick = loadAnalysis; $("trainLatencyButton").onclick = () => trainModel("latency_prediction", "trainLatencyButton"); $("trainRootCauseButton").onclick = () => trainModel("root_cause", "trainRootCauseButton"); $("modelPredictButton").onclick = predictModel; $("runSourceAnalysisButton").onclick = runSourceAnalysis; $("retrieveCodeKnowledgeButton").onclick = retrieveCodeKnowledge; $("differentialRefresh").onclick = loadAnalysis; $("agentRefresh").onclick = loadAgent; $("agentOptimizeButton").onclick = runAgent; $("mineExperiencesButton").onclick = mineExperiences; window.onresize = () => { if (["overview", "workloads"].includes(state.currentView)) drawChart(state.metrics || []); }; }
function schedule() { clearTimeout(state.timer); if (state.autoRefresh) state.timer = setTimeout(async () => { await refreshDashboard(); if (state.currentView === "incidents") await loadIncidents(); if (state.currentView === "actions") await loadActions(); if (state.currentView === "observability") await loadObservability(); if (state.currentView === "analysis") await loadAnalysis(); if (state.currentView === "agent") await loadAgent(); schedule(); }, 5000); }
const viewFromHash = () => {
const view = window.location.hash.replace(/^#/, "");
return document.getElementById(`view-${view}`) ? view : "overview";
};
async function init() { const exportButton = $("exportEvidenceButton"); if (exportButton) exportButton.onclick = exportEvidence; const cleanButton = $("cleanDemoButton"); if (cleanButton) cleanButton.onclick = cleanDemoState; bind(); await refreshDashboard(); setView(viewFromHash()); window.addEventListener("hashchange", () => setView(viewFromHash())); schedule(); }
document.addEventListener("DOMContentLoaded", init);
})();
